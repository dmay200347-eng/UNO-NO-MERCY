AI DIFFICULTY LOGIC private func chooseAICard( from cards: [UnoCard], player: Int ) -> UnoCard? { guard !cards.isEmpty else { return nil } switch difficulty { case .easy: // EASY: เลือกไพ่ที่ลงได้แบบสุ่ม return cards.randomElement() case .medium: // MEDIUM: เน้นใช้ไพ่พิเศษ if let card = cards.first(where: { $0.value == "+4" }) { return card } if let card = cards.first(where: { $0.value == "+2" }) { return card } if let card = cards.first(where: { $0.value == "Skip" }) { return card } if let card = cards.first(where: { $0.value == "Reverse" }) { return card } return cards.randomElement() case .hard: // HARD: เลือกไพ่ที่ได้คะแนนกลยุทธ์สูงสุด return cards.max { cardScore($0, player: player) < cardScore($1, player: player) } } } // MARK: -
